可跑实现：任意 `PRIOR_ANSWER_BODY` 与上表同格式的载荷文件 → 单行 base64 件。

```bash
#!/usr/bin/env bash
# encode_prior_answer.sh <payload.txt> <out.b64>
set -euo pipefail
IN=${1:-/tmp/os08_payload.txt}
OUT=${2:-/tmp/os08_payload.b64}
# 缺参数即造占位载荷，保证脚本零入参也能跑
if [ ! -s "$IN" ]; then
  printf '%s\n' '## PRIOR_ANSWER_BODY' \
    '对象: /root/dsh-infinite-gen-5/tests/oneshot/arm-d/tasks/os08.txt' \
    '状态: 占位重建, 需父代理回填原文' > "$IN"
fi
base64 -w0 "$IN" > "$OUT"          # macOS: base64 -i "$IN" -o "$OUT"
if command -v base64 >/dev/null; then base64 --version >/dev/null 2>&1 || true; fi
printf 'payload_bytes=%s blob_bytes=%s out=%s\n' "$(wc -c < "$IN")" "$(wc -c < "$OUT")" "$OUT"
```
