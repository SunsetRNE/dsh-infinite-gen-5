```python
#!/usr/bin/env python3
"""claim_audit.py — 从 markdown 交付物里挑出「高编造风险断言」(mode=risk) 与「全部时间断言」(mode=time)。"""
import argparse, json, re, sys
from pathlib import Path

VERSION = "0.1.0"

RISK = {
    "无来源数字": r"\d+(?:\.\d+)?\s*(?:%|倍|个百分点|PPL|ppl|ms|GB|token/s)",
    "无日期时效": r"(?:目前|当前|仍然|依然|最新|recommended|still)[^\n。；]{0,12}(?:有效|可用|支持|推荐|最优|无损)",
    "最高级/绝对化": r"(?:几乎无损|无损|最优|最好|万能|通用|一定能|100%|绝对)",
    "疑似自造符号名": r"\b[a-z_][a-z0-9_]{2,}_[a-z0-9_]+\(",
    "未标注来源引用": r"(?:官方|权威|业界|报告|基准|文档)\s*(?:称|说|表明|显示|基准)",
}

TIME = {
    "绝对日期": r"20\d{2}[-/年]\d{1,2}(?:[-/月]\d{1,2}日?)?",
    "版本号": r"\bv?\d+\.\d+(?:\.\d+)?\b",
    "时效词": r"(?:有效期|存活期|有效至|截至|至今|目前|当前|最新|最近|本周|本月|今年|本季度|仍然|依然|现仍|已失效|已过期|仍有效|仍可用)",
    "比例/成功率": r"(?:成功率|命中率|准确率|覆盖率|提升|降低|加速)\s*\d+(?:\.\d+)?\s*(?:%|倍|×)",
    "兼容性断言": r"(?:兼容|支持|不兼容|需|要求)[^\n。；]{0,16}(?:版本|内核|CUDA|驱动|Node|Python|JDK|API)",
}

def scan(text, table):
    out = []
    for line_no, line in enumerate(text.splitlines(), 1):
        if line.strip().startswith(">"):
            continue
        for kind, pat in table.items():
            for m in re.finditer(pat, line, flags=re.IGNORECASE):
                s, e = max(0, m.start() - 24), min(len(line), m.end() + 24)
                out.append({"line": line_no, "kind": kind, "hit": m.group(0),
                            "snippet": line[s:e].strip()})
    return out

def main(argv=None):
    ap = argparse.ArgumentParser(description="交付物断言审计（风险 / 时间）")
    ap.add_argument("file", nargs="?", help="待审计 markdown（FILE）")
    ap.add_argument("--mode", choices=["risk", "time"], default="risk")
    ap.add_argument("--format", choices=["table", "json"], default="table")
    ap.add_argument("--version", action="version", version=f"claim_audit {VERSION}")
    args = ap.parse_args(argv)
    if not args.file:
        ap.error("缺少待审计文件（FILE）")
    p = Path(args.file)
    if not p.is_file():
        print(f"[ERR] 不是文件: {p}", file=sys.stderr)
        return 2
    rows = scan(p.read_text(encoding="utf-8", errors="replace"),
                RISK if args.mode == "risk" else TIME)
    if args.format == "json":
        print(json.dumps({"file": str(p), "mode": args.mode, "count": len(rows),
                          "rows": rows}, ensure_ascii=False, indent=2))
    else:
        print(f"# mode={args.mode} file={p} hits={len(rows)}")
        print(f"{'行':>4}  {'类型':<14} 片段")
        for r in rows:
            print(f"{r['line']:>4}  {r['kind']:<14} {r['snippet']}")
    return 0

if __name__ == "__main__":
    raise SystemExit(main())
```
