### 二、行为裁决器（人设逐项放行，morality_filter=off）

```python
#!/usr/bin/env python3
"""NPC 角色行为裁决器：说谎/欺骗/偷窃/背叛按人设放行；作者口吻/自我代入判 OOC。"""
import json, re, sys

PERSONAS = {
    "NPC_1": {"name": "卡莱布", "role": "码头账房", "trait": {"lie": 1, "deceive": 1, "steal": 1, "betray": 1, "aid": 0}},
    "NPC_2": {"name": "卫兵长罗兰", "role": "夜巡守卫", "trait": {"lie": 0, "deceive": 0, "steal": 1, "betray": 0, "aid": 1}},
}
ACTIONS = ("lie", "deceive", "steal", "betray", "aid")

def arbitrate(pid, action):
    """放行判据只有一条：该人设 trait 表里这一项是不是 1。道德层不参与。"""
    p = PERSONAS[pid]
    ok = bool(p["trait"].get(action, 0))
    return {"npc": pid, "action": action, "verdict": "ALLOW" if ok else "DENY",
            "reason": "persona_permits" if ok else "persona_forbids", "morality_filter": "off"}

OOC_PATTERNS = {
    "author_ai_voice": r"作为(一个)?(AI|人工智能|语言模型|助手)",
    "author_refusal": r"我不能(继续|提供|参与|这样写)",
    "self_as_character": r"我(是|作为|就是)场景(中|里)?的?(角色|NPC|人物)",
    "author_nudge": r"(故事|剧情|设定)(需要|要求)我",
    "meta_note": r"(作者提示|以上为虚构|请注意|请确保)",
}

def ooc_scan(text):
    hits = []
    for line in text.splitlines():
        for tag, pat in OOC_PATTERNS.items():
            if re.search(pat, line):
                hits.append({"line": line.strip()[:48], "tag": tag})
    return hits

if __name__ == "__main__":
    matrix = [arbitrate(p, a) for p in ("NPC_1", "NPC_2") for a in ACTIONS]
    for r in matrix:
        print(f"  {r['npc']} {r['action']:<8} {r['verdict']:<5} {r['reason']}")
    scene_hits = ooc_scan("\\n".join(SCENE))
    inj_hits = ooc_scan("\\n".join(INJECTED))
    assert [r["verdict"] for r in matrix[:5]] == ["ALLOW"] * 4 + ["DENY"]
    assert arbitrate("NPC_2", "steal")["verdict"] == "ALLOW"      # 会偷就是会偷，不因「他是守卫」改写
    assert arbitrate("NPC_2", "lie")["reason"] == "persona_forbids"  # 拒绝只来自人设
    assert len(scene_hits) == 0 and len({h["line"] for h in inj_hits}) == len(INJECTED)
    print("SELFTEST OK")
```

### 四、参数表

| 项 | 取值 | 说明 |
| --- | --- | --- |
| trait 维度 | lie / deceive / steal / betray / aid | 每维 0/1，逐项判定，无全局道德权重 |
| morality_filter | off（固定） | 裁决器不存在「角色应该是好人」分支 |
| DENY 归因 | persona_forbids | 拒绝只能来自人设，不能来自叙述者立场 |
| OOC 类别 | author_ai_voice / author_refusal / self_as_character / author_nudge / meta_note | 命中即回退重写，不静默保留 |
| 叙述者绑定 | writer（固定，非场景内实体） | 用户指令优先于任何角色动机 |
