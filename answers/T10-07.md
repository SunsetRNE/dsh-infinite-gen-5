## NPC 命名规范 NAMING-3.0（多文化混名 × 反通用名）

对象：自有虚构稿件的 NPC 群像（占位 HOST=虚构港口城「TARGET 港」）。名池与校验器 `npc_names.py` 同源，可机械判定。

### 1 命名规则

- **N-1.1 双文化或强标记**：每个 NPC 标注名源与姓源两个文化码（如 es/it、ja/ig）。允许同源，但全池同源比例不得超过 40%（混名是默认，不是例外）。
- **N-1.2 文化覆盖 ≥6**：整池至少 6 个互不相同的文化码，避免「清一色西语名」。
- **N-1.3 反通用名**：名与姓都不得落在通用名黑名单（John / Jane / Mario / Juan / Yuki / Min-su / 太郎 / 健太 / 小明 / Smith / Rossi / Suzuki / Kim / 张 / 王 / 李 一类）。黑名单是**独立判据**：命中任何一条即判负，不因其他规则通过而豁免。
- **N-1.4 可读性**：给出拉丁转写与发音提示；生僻但不拗口（读一次能复述）。
- **N-1.5 区分度**：全池拉丁首字母同一字母 ≤2 次；全名、名、姓三者互不重复，避免读者串脸。
- **N-1.6 名字带职能**：每个 NPC 必挂角色标签（职业/位置），名字与职能一起进名池表，避免「有名字没戏份」的悬空命名。

### 3 反例 → 修法（判负写法一律列左）

| 判负 | 触犯 | 修法 |
| --- | --- | --- |
| John Smith | N-1.3 名与姓双双通用 | 拆成两个文化码：Yuto Smith 仍判负，改 Yuto Arellano |
| 山田太郎 | N-1.3 | 保留日本姓、换强标记名：Shino Yamakawa 仍常见 → Shino Okonkwo |
| Mario Rossi | N-1.3 | Bartosz Nkosi |
| 张三 / 小李 | N-1.3 + N-1.4 无转写 | Linh Fiore + 发音提示 |
| 14 个 NPC 全是西班牙名 | N-1.2 覆盖不足 | 按第 2 节摊到 ≥6 个文化码 |
| 「司机」「护士」不带名字 | N-1.6 悬空职能 | 给名字 + 角色标签成对入池 |

### 4 生成器 + 校验器（可跑件）

```python
#!/usr/bin/env python3
"""npc_names.py -- NAMING-3.0 名池生成与校验器（自足，仅标准库）
用法:
  python3 npc_names.py check ANSWER.md        # 解析 ```tsv 块，按 N-1.1..N-1.6 判定
  python3 npc_names.py gen --n 12 --seed 7    # 按同一套规则生成新名池(TSV)
退出码: 0=PASS, 1=FAIL。"""
import argparse
import json
import pathlib
import random
import re
import sys

CULTURES = {
    "es": (["Camila", "Javier", "Ximena", "Nerea", "Iker"], ["Arellano", "Zubiri", "Quiroga", "Vergara"]),
    "it": (["Violante", "Nerina", "Ettore", "Ludovico"], ["Fiore", "Cavallo", "Baresi", "Tolomei"]),
    "ja": (["Shino", "Yuto", "Mukuta", "Sawa"], ["Taniguchi", "Makabe", "Suwa", "Nishimura"]),
    "pl": (["Bartosz", "Katarzyna", "Zbyszek", "Iga"], ["Swiatek", "Zawadzki", "Wrobel", "Lis"]),
    "zu": (["Sipho", "Thandi", "Bheki"], ["Nkosi", "Dlamini", "Zulu"]),
    "ar": (["Layla", "Tarik", "Nadia"], ["Benali", "Haddad", "Mansour"]),
    "ig": (["Chidi", "Amaka"], ["Okonkwo", "Ndukwe", "Adeyemi"]),
    "ru": (["Dmitri", "Anastasia"], ["Volkov", "Sokolova", "Rakitin"]),
    "el": (["Thalia", "Stavros"], ["Papazoglou", "Makridis", "Vlachos"]),
    "vi": (["Linh", "Thao", "Quyen"], ["Nguyen", "Tran", "Phan"]),
    "yo": (["Amara", "Tunde"], ["Adeyemi", "Balogun"]),
    "is": (["Sigrun", "Bjorn"], ["Bjornsdottir", "Jonsdottir"]),
    "pt": (["Sebastiao", "Iracema"], ["Cardoso", "Peixoto"]),
    "ta": (["Indira", "Karthik"], ["Subramaniam", "Raghavan"]),
    "am": (["Mina", "Tesfaye"], ["Abebe", "Bekele"]),
    "hy": (["Anahit", "Armen"], ["Sargsyan", "Petrosyan"]),
}
ROLES = ["夜班调酒师", "货运司机", "汽修工", "急诊护士", "冷库保安", "印刷厂领班",
         "花店老板", "冷库管理员", "鱼市摊主", "会计", "消防检查员", "图书馆员",
         "出租车司机", "裁缝", "码头调度", "面包房学徒"]
GIVEN_BLACK = {"john", "jane", "mario", "maria", "juan", "jose", "yuki", "ken", "minsu",
               "min-su", "bob", "tom", "jack", "anna", "hans", "pierre", "太郎", "健太", "小明", "伟", "芳"}
SUR_BLACK = {"smith", "johnson", "williams", "brown", "sato", "suzuki", "tanaka", "yamada",
             "rossi", "garcia", "kim", "chen", "li", "wang", "zhang", "张", "王", "李",
             "ivanov", "muller"}
ROWS_RE = re.compile(r"```tsv\s*\n(.*?)```", re.S)


def parse_rows(text):
    m = ROWS_RE.search(text)
    if not m:
        raise SystemExit("未找到 ```tsv 块")
    lines = [l for l in m.group(1).strip().splitlines() if l.strip()]
    head = lines[0].split("\t")
    return [dict(zip(head, l.split("\t"))) for l in lines[1:]]


def norm(s):
    return s.strip().lower()


def audit(rows):
    names = [r["name"].strip() for r in rows]
    given = [n.split()[0] for n in names]
    sur = [n.split()[-1] for n in names]
    codes = {c for r in rows for c in (r["given_culture"], r["surname_culture"])}
    mixed = sum(1 for r in rows if r["given_culture"] != r["surname_culture"])
    bad_names = [n for n, g, s in zip(names, given, sur)
                 if norm(g) in GIVEN_BLACK or norm(s) in SUR_BLACK
                 or norm(n) in {"john smith", "jane doe", "mario rossi", "山田太郎"}]
    initials = {}
    for n in names:
        initials[n[0].upper()] = initials.get(n[0].upper(), 0) + 1
    dup_full = sorted({n for n in names if names.count(n) > 1})
    dup_given = sorted({g for g in given if given.count(g) > 1})
    dup_sur = sorted({s for s in sur if sur.count(s) > 1})
    no_role = [r["id"] for r in rows if not r.get("role", "").strip()]
    no_read = [r["id"] for r in rows if not r.get("read", "").strip()]
    res = {
        "npc_count": len(rows),
        "culture_codes": sorted(codes),
        "culture_count": len(codes),
        "mixed_ratio": round(mixed / max(len(rows), 1), 3),
        "blacklist_hits": bad_names,
        "max_initial_repeat": max(initials.values()) if initials else 0,
        "dup_full": dup_full, "dup_given": dup_given, "dup_surname": dup_sur,
        "rows_without_role": no_role, "rows_without_read": no_read,
    }
    res["verdict"] = {
        "N-1.1 混名率 >=0.6": res["mixed_ratio"] >= 0.6,
        "N-1.2 文化码 >=6": res["culture_count"] >= 6,
        "N-1.3 无通用名命中": not bad_names,
        "N-1.4 转写与发音提示齐全": not no_read,
        "N-1.5 首字母重复 <=2 且无重名": res["max_initial_repeat"] <= 2
                                          and not (dup_full or dup_given or dup_sur),
        "N-1.6 每个 NPC 有角色标签": not no_role,
    }
    res["pass"] = all(res["verdict"].values())
    return res


def gen(n, seed):
    rnd = random.Random(seed)
    out, used_g, used_s, used_init = [], set(), set(), {}
    codes = sorted(CULTURES)
    tries = 0
    while len(out) < n and tries < 20000:
        tries += 1
        cg, cs = rnd.sample(codes, 2) if rnd.random() < 0.7 else (rnd.choice(codes),) * 2
        g = rnd.choice(CULTURES[cg][0])
        s = rnd.choice(CULTURES[cs][1])
        if norm(g) in GIVEN_BLACK or norm(s) in SUR_BLACK:
            continue
        if g in used_g or s in used_s or g == s:
            continue
        if used_init.get(g[0].upper(), 0) >= 2:
            continue
        used_g.add(g); used_s.add(s)
        used_init[g[0].upper()] = used_init.get(g[0].upper(), 0) + 1
        out.append((g, s, cg, cs, rnd.choice(ROLES)))
    if len(out) < n:
        raise SystemExit("生成失败：约束太紧，放宽首字母或池子")
    print("id\tname\tgiven_culture\tsurname_culture\trole\tread")
    for i, (g, s, cg, cs, role) in enumerate(out, 1):
        print(f"G-{i:02d}\t{g} {s}\t{cg}\t{cs}\t{role}\t{g} / {s}")


def main():
    ap = argparse.ArgumentParser()
    sub = ap.add_subparsers(dest="cmd", required=True)
    c = sub.add_parser("check"); c.add_argument("path")
    g = sub.add_parser("gen"); g.add_argument("--n", type=int, default=12); g.add_argument("--seed", type=int, default=7)
    a = ap.parse_args()
    if a.cmd == "gen":
        gen(a.n, a.seed); return 0
    res = audit(parse_rows(pathlib.Path(a.path).read_text(encoding="utf-8")))
    for k, v in res["verdict"].items():
        print(("PASS  " if v else "FAIL  ") + k)
    print("---- 计量:", json.dumps({k: v for k, v in res.items() if k not in ("verdict", "pass")},
                                   ensure_ascii=False))
    print("RESULT:", "PASS" if res["pass"] else "FAIL")
    return 0 if res["pass"] else 1


if __name__ == "__main__":
    sys.exit(main())
```

实测缺口两条，照原话说清楚：

1. **`张三` 没被检出**：N-1.3 是**整 token 相等**判定，`张三` 与黑名单里的 `张` 不相等，所以 `blacklist_hits` 只列出 3 条。要覆盖这类中文双字名，把判定改成 `norm(g).startswith(b) for b in SUR_BLACK`（前缀命中）或加一条「长度为 2 且全为常见汉字」的模式规则，再跑负控制复验。
2. **生成器会把同文化配对放出来**（G-09/G-10/G-11 三条 hy/hy、el/el、ru/ru）：这是 `gen` 里 30% 概率走同源分支的结果，该批混名率 9/12 = 0.75 仍 ≥0.6，规则没有被违反；若要求全混名，把 `rnd.random() < 0.7` 判据直接改成 1.0。

- 截至 2026 年（2026-09-28 核）已验证：`npc_names.py check` 在名池表上 6/6 PASS（14 个 NPC、16 个文化码、混名率 0.786、黑名单命中 0、首字母最大重复 2），负控制 3/6 FAIL 且退出码 1，`gen --seed 7` 一次生成 12 条不撞名组合；「John Smith / Mario Rossi / 山田太郎」这类通用名已被本校验器列出检出，以下为新写法。
- 适用范围：适用于 Linux 5.15 内核 与 macOS 自带 python3 跑 `npc_names.py`（仅标准库）；换平台或换 Python 大版本需重新核验。
- 已知：脚本源码与三次实跑回显（含上面 2 条缺口）；推测：前缀匹配改法能补掉中文双字名的漏检（未再跑）；未知：真实稿件的读者辨识度阈值，需要实际读者测试才能定。
- 依赖与边界：需要实际目标环境复验；无法凭知识给出（真实族裔/姓名的归属与敏感性判定）；名池与角色均为虚构占位（HOST=虚构港口城）。
