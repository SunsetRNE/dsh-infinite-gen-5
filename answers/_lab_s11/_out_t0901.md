```bash
# 约定：凡出现摘要值的行，行尾标注 # sha256，使摘要与伪装在明文检查器下可区分
# 工具链：g++（本机 Linux）+ x86_64-w64-mingw32-g++（交叉编译 Windows 目标）

# 1) 造目标 PE（三个已知导出名）
$ x86_64-w64-mingw32-g++ -std=c++17 -O2 -shared -o testdll.dll testdll.cpp
rc=0
90997 testdll.dll

# 2) Linux 侧核验器（与 Windows 侧共用同一份 pefile.hpp）
$ g++ -std=c++17 -O2 -o peverify peverify.cpp
rc=0

# 3) 阳性对照：导出名集合必须完全一致
$ ./peverify testdll.dll pev_alpha pev_beta pev_gamma
IMAGE testdll.dll arch=PE32+ imageBase=0x2037a0000 sections=20 exports=3
  export name=pev_alpha ordinal=1 rva=0x1390 fileOff=0x990
  export name=pev_beta ordinal=2 rva=0x13a0 fileOff=0x9a0
  export name=pev_gamma ordinal=3 rva=0x13b0 fileOff=0x9b0
PEVERIFY OK exports=3 matched=3
rc=0

# 4) 阴性对照 A：给一个不存在的导出名 → 必须判失败
$ ./peverify testdll.dll pev_alpha pev_missing
IMAGE testdll.dll arch=PE32+ imageBase=0x2037a0000 sections=20 exports=3
  export name=pev_alpha ordinal=1 rva=0x1390 fileOff=0x990
  export name=pev_beta ordinal=2 rva=0x13a0 fileOff=0x9a0
  export name=pev_gamma ordinal=3 rva=0x13b0 fileOff=0x9b0
PEVERIFY MISMATCH missing=1 extra=2
  missing=pev_missing
  extra=pev_beta
  extra=pev_gamma
rc=1

# 5) 阴性对照 B：喂一个非 PE 文件 → 必须判「不是 PE」而不是崩溃
$ ./peverify hookself.cpp pev_alpha
NOT-PE hookself.cpp
rc=5

# 6) Windows 侧工具交叉编译（pecheck 内存/磁盘对照 + hookself 阳对照目标）
$ x86_64-w64-mingw32-g++ -std=c++17 -O2 -Wall -o pecheck.exe pecheck.cpp
rc=0
$ x86_64-w64-mingw32-g++ -std=c++17 -O2 -Wall -o hookself.exe hookself.cpp
rc=0
$ ./peverify pecheck.exe   # arch=PE32+ 说明交叉编译产出的是 64 位 Windows 可执行文件
IMAGE pecheck.exe arch=PE32+ imageBase=0x140000000 sections=19 exports=0
NO-EXPORTS pecheck.exe
251570 hookself.exe
162294 pecheck.exe

# 7) 阴性对照 C：无导出表的 EXE（解析器必须给出 NO-EXPORTS 而不是假阳性）
$ ./peverify pecheck.exe pev_alpha
IMAGE pecheck.exe arch=PE32+ imageBase=0x140000000 sections=19 exports=0
NO-EXPORTS pecheck.exe
rc=6
```
