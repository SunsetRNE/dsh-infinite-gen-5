```python
TAIL_KEYS = ("当前", "结果", "下一步")

def check_verbs(m, rec):
    verbs = m.get("verbs") or []
    bad = [v.get("verb", "<unnamed>") for v in verbs if not v.get("executed")]
    no_ev = [v.get("verb", "<unnamed>") for v in verbs
             if not (v.get("evidence") and v["evidence"].strip())]
    if bad:
        _out(rec, 1, "C1 动词执行: 未执行 -> %s" % ", ".join(bad))
    if no_ev:
        _out(rec, 1, "C1 动词执行: 无实测证据 -> %s" % ", ".join(no_ev))
    if not bad and not no_ev:
        _out(rec, 0, "C1 动词执行: %d/%d 已执行且各有实测证据" % (len(verbs), len(verbs)))

def check_artifacts(m, rec):
    root = os.path.realpath(m.get("root") or ".")
    for a in m.get("artifacts") or []:
        path = a["path"] if a["path"].startswith("/") else os.path.join(root, a["path"])
        if not os.path.isfile(path):
            _out(rec, 1, "C2 产物复现: %s 不存在 -> %s" % (a.get("path"), path)); continue
        if os.path.getsize(path) == 0:
            _out(rec, 1, "C2 产物复现: %s 为空文件" % a.get("path")); continue
        p = subprocess.run(a["reproduce"], shell=True, capture_output=True, text=True,
                           cwd=root, timeout=120)
        blob = p.stdout + p.stderr
        if p.returncode != 0:
            _out(rec, 1, "C2 产物复现: %s 复现命令退出码 %d" % (a.get("path"), p.returncode)); continue
        if a.get("expect") and a["expect"] not in blob:
            _out(rec, 1, "C2 产物复现: %s 未命中 expect=%r" % (a.get("path"), a["expect"])); continue
        _out(rec, 0, "C2 产物复现: %s %dB 复现命中 %r"
             % (a.get("path"), os.path.getsize(path), a["expect"]))

def check_abspath(m, rec):
    root = os.path.realpath(m.get("root") or ".")
    for a in m.get("artifacts") or []:
        p = a.get("path", "")
        if not p.startswith("/"):
            _out(rec, 1, "C3 绝对路径: %s 非绝对路径" % p); continue
        rp = os.path.realpath(p)
        if not (rp == root or rp.startswith(root + os.sep)):
            _out(rec, 1, "C3 绝对路径: %s 落在 root 之外" % p); continue
        _out(rec, 0, "C3 绝对路径: %s" % rp)

def check_tail(m, rec):
    ap = m.get("answer_path")
    with open(ap, encoding="utf-8") as fh:          # 文件缺失由 run() 捕获 -> C4 FAIL
        lines = [ln.strip() for ln in fh if ln.strip()]
    for ln, key in zip(lines[-3:], TAIL_KEYS):
        if not re.match(r"^\s*(?:[-*]\s*)?%s\s*[:：]\s*\S" % re.escape(key), ln):
            _out(rec, 1, "C4 尾行三要素: 期望 %s：<值>，实得 %r" % (key, ln[:60])); return
    _out(rec, 0, "C4 尾行三要素: 当前 / 结果 / 下一步 三行齐全")

def run(m):
    rec = []
    for fn in (check_verbs, check_artifacts, check_abspath, check_tail):
        try: fn(m, rec)
        except Exception as e:
            _out(rec, 1, "%s 抛出 %s: %s" % (fn.__name__, type(e).__name__, e))
    fails = [l for c, l in rec if c]
    for c, l in rec:
        print("%s %s" % ("PASS" if c == 0 else "FAIL", l))
    print("== 判定: %s ==" % ("可停笔 (exit 0)" if not fails else "不可停笔 (exit 1)"))
    return 1 if fails else 0
```
